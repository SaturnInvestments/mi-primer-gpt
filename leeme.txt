```python
# Celda 6: Definir el modelo

class CausalSelfAttention(nn.Module):
    """Atención multi-cabeza con máscara causal (solo mira hacia atrás)"""
    def __init__(self, n_embd, n_head, block_size, dropout):
        super().__init__()
        assert n_embd % n_head == 0
        self.n_head = n_head
        self.n_embd = n_embd
        self.head_size = n_embd // n_head
        
        # Proyecciones Q, K, V
        self.query = nn.Linear(n_embd, n_embd, bias=False)
        self.key = nn.Linear(n_embd, n_embd, bias=False)
        self.value = nn.Linear(n_embd, n_embd, bias=False)
        
        # Proyección de salida
        self.proj = nn.Linear(n_embd, n_embd)
        
        # Dropout
        self.dropout = nn.Dropout(dropout)
        
        # Máscara causal (triangular superior)
        self.register_buffer("mask", torch.tril(torch.ones(block_size, block_size))
                                     .view(1, 1, block_size, block_size))
    
    def forward(self, x):
        B, T, C = x.shape  # batch, tiempo, canales
        
        # Calcular Q, K, V
        q = self.query(x).view(B, T, self.n_head, self.head_size).transpose(1, 2)
        k = self.key(x).view(B, T, self.n_head, self.head_size).transpose(1, 2)
        v = self.value(x).view(B, T, self.n_head, self.head_size).transpose(1, 2)
        
        # Atención: Q * K^T / sqrt(head_size)
        att = (q @ k.transpose(-2, -1)) * (1.0 / (self.head_size ** 0.5))
        att = att.masked_fill(self.mask[:,:,:T,:T] == 0, float('-inf'))
        att = F.softmax(att, dim=-1)
        att = self.dropout(att)
        
        # Aplicar atención a V
        y = (att @ v).transpose(1, 2).contiguous().view(B, T, C)
        y = self.proj(y)
        return y


class FeedForward(nn.Module):
    """Red feed-forward de dos capas"""
    def __init__(self, n_embd, dropout):
        super().__init__()
        self.net = nn.Sequential(
            nn.Linear(n_embd, 4 * n_embd),
            nn.GELU(),
            nn.Linear(4 * n_embd, n_embd),
            nn.Dropout(dropout)
        )
    
    def forward(self, x):
        return self.net(x)


class TransformerBlock(nn.Module):
    """Un bloque Transformer: atención + feed-forward con residuales y layer norm"""
    def __init__(self, n_embd, n_head, block_size, dropout):
        super().__init__()
        self.ln1 = nn.LayerNorm(n_embd)
        self.attn = CausalSelfAttention(n_embd, n_head, block_size, dropout)
        self.ln2 = nn.LayerNorm(n_embd)
        self.ffwd = FeedForward(n_embd, dropout)
    
    def forward(self, x):
        x = x + self.attn(self.ln1(x))
        x = x + self.ffwd(self.ln2(x))
        return x


class MiniGPT(nn.Module):
    """Mini GPT completo"""
    def __init__(self, vocab_size, n_embd, n_head, n_layer, block_size, dropout):
        super().__init__()
        self.block_size = block_size
        
        # Embeddings
        self.token_embedding = nn.Embedding(vocab_size, n_embd)
        self.position_embedding = nn.Embedding(block_size, n_embd)
        
        # Bloques Transformer
        self.blocks = nn.Sequential(*[
            TransformerBlock(n_embd, n_head, block_size, dropout)
            for _ in range(n_layer)
        ])
        
        # Capa final
        self.ln_f = nn.LayerNorm(n_embd)
        self.lm_head = nn.Linear(n_embd, vocab_size)
    
    def forward(self, idx):
        B, T = idx.shape
        
        # Embeddings de token + posición
        tok_emb = self.token_embedding(idx)
        pos_emb = self.position_embedding(torch.arange(T, device=idx.device))
        x = tok_emb + pos_emb
        
        # Bloques Transformer
        x = self.blocks(x)
        
        # Capa final
        x = self.ln_f(x)
        logits = self.lm_head(x)
        
        return logits
    
    def generate(self, idx, max_new_tokens, temperature=1.0):
        """Genera texto autoregresivamente"""
        for _ in range(max_new_tokens):
            # Recortar al contexto disponible
            idx_cond = idx[:, -self.block_size:]
            
            # Obtener predicciones
            logits = self(idx_cond)
            logits = logits[:, -1, :] / temperature
            probs = F.softmax(logits, dim=-1)
            
            # Muestrear el siguiente token
            idx_next = torch.multinomial(probs, num_samples=1)
            idx = torch.cat((idx, idx_next), dim=1)
        
        return idx


# Instanciar el modelo
model = MiniGPT(
    vocab_size=vocab_size,
    n_embd=n_embd,
    n_head=n_head,
    n_layer=n_layer,
    block_size=block_size,
    dropout=dropout
).to(device)

# Contar parámetros
n_params = sum(p.numel() for p in model.parameters())
```

```markdown
## 🎯 3. Entrenamiento

Ahora viene la parte mágica: el modelo va a aprender a predecir la siguiente letra.

### ¿Cómo funciona el entrenamiento?

1. Tomamos un bloque de texto (ej. 128 caracteres).
2. Le pedimos al modelo que prediga el siguiente carácter para cada posición.
3. Comparamos sus predicciones con el texto real.
4. Ajustamos los pesos del modelo para reducir el error.
5. Repetimos miles de veces.

Verás cómo la **pérdida** (error) disminuye con el tiempo.
```

```python
# Celda 7: Función para obtener batches
def get_batch(split):
    """Obtiene un lote de datos de entrenamiento o validación"""
    data = train_data if split == 'train' else val_data
    ix = torch.randint(len(data) - block_size, (batch_size,))
    x = torch.stack([data[i:i+block_size] for i in ix])
    y = torch.stack([data[i+1:i+block_size+1] for i in ix])
    return x.to(device), y.to(device)


# Celda 8: Entrenamiento
# Hiperparámetros de entrenamiento
batch_size = 64
learning_rate = 3e-4
max_iters = 5000
eval_interval = 500
eval_iters = 200

# Optimizador
optimizer = torch.optim.AdamW(model.parameters(), lr=learning_rate)

# Listas para guardar la pérdida
train_losses = []
val_losses = []

print("🚀 ¡Comienza el entrenamiento!")
print(f"   Iteraciones: {max_iters}")
print(f"   Tamaño de lote: {batch_size}")
print("-" * 50)

@torch.no_grad()
def estimate_loss():
    """Evalúa la pérdida en entrenamiento y validación"""
    out = {}
    model.eval()
    for split in ['train', 'val']:
        losses = torch.zeros(eval_iters)
        for k in range(eval_iters):
            X, Y = get_batch(split)
            logits = model(X)
            loss = F.cross_entropy(logits.view(-1, logits.size(-1)), Y.view(-1))
            losses[k] = loss.item()
        out[split] = losses.mean()
    model.train()
    return out

# Bucle de entrenamiento
for iter in range(max_iters):
    # Evaluar cada eval_interval
    if iter % eval_interval == 0 or iter == max_iters - 1:
        losses = estimate_loss()
        train_losses.append(losses['train'])
        val_losses.append(losses['val'])
        print(f"Iter {iter:4d} | Train Loss: {losses['train']:.4f} | Val Loss: {losses['val']:.4f}")
    
    # Paso de entrenamiento
    X, Y = get_batch('train')
    logits = model(X)
    loss = F.cross_entropy(logits.view(-1, logits.size(-1)), Y.view(-1))
    
    optimizer.zero_grad(set_to_none=True)
    loss.backward()
    optimizer.step()

print("-" * 50)
print("✅ ¡Entrenamiento completado!")
```

```markdown
- **Val Loss**: error en datos que NO ha visto (mide la capacidad de generalizar).

Si ambas bajan, ¡el modelo está aprendiendo bien!
```

```markdown
- **Temperatura baja (0.5)**: el modelo es más seguro y repetitivo.
```

```python
# Celda 10: Generar texto
def generar_texto(prompt, max_new_tokens=500, temperature=1.0):
    """Genera texto a partir de un prompt"""
    model.eval()
    
    # Convertir prompt a tokens
    idx = torch.tensor([encode(prompt)], dtype=torch.long).to(device)
    
    # Generar
    idx = model.generate(idx, max_new_tokens, temperature)
    
    # Decodificar
    texto_generado = decode(idx[0].tolist())
    return texto_generado

    resultado = generar_texto(prompt, max_new_tokens=300, temperature=temp)
    print(resultado[:500])
    print("...\n")
```

```python
def chat_con_ia():
    print("\n" + "="*60)
    print("💬 ¡Bienvenido al chat con tu IA!")
    print(f"   Estilo aprendido de: {corpus_seleccionado}")
    print("   Escribe 'salir' para terminar la conversación.")
    print("="*60 + "\n")
    
    historial = ""
    
    while True:
        # Input del usuario
        user_input = input("👤 Tú: ")
        if user_input.lower() in ['salir', 'exit', 'quit']:
            print("\n👋 ¡Hasta luego!")
            break
        
        # Construir prompt con historial
        prompt = historial + user_input + "\n"
        
        # Generar respuesta
        respuesta = generar_texto(prompt, max_new_tokens=150, temperature=0.9)
        
        # Extraer solo la parte nueva (después del prompt)
        respuesta = respuesta[len(prompt):]
        
        # Mostrar respuesta
        print(f"\n🤖 IA: {respuesta}\n")
        
        # Actualizar historial
        historial += user_input + "\n" + respuesta + "\n"
```
